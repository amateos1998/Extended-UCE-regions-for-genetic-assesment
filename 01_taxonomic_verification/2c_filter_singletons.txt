#!/bin/bash
#SBATCH --job-name=single_filter     # Job name
#SBATCH --partition=cpuq   # Specify the partition you want to use
#SBATCH --cpus-per-task=2                 # Number of CPU cores per task
#SBATCH --time=48:00:00                   # Time limit hrs:min:sec
#SBATCH --mem=10G                          # Memory per node
#SBATCH --output=/your/path/ # Standard output log
#SBATCH --error=/your/path/

###THIS SCRIPT filters out singletons from loci fastas from UCE extraction

# Activate the conda environment where Phyluce is installed
source /your/path/miniconda3/etc/profile.d/conda.sh
conda activate phyluce-1.7.3


# this is done from within the directory where your loci fastas are

mkdir -p /your/path/exploded-fastas/singletons

# this just grabs the fiels where there is only 1 “>” – so only one fasta seq in the file and move them to the new folder “singletons”

grep -c ">" *.fasta | awk -F ':' '$2==1{print $1}' | xargs -I {} mv {} /your/path/exploded-fastas/singletons
